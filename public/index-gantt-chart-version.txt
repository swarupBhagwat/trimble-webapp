function stats() {
  const counts = PHASES.map(() => 0);
  const recs = Object.values(progress);
  recs.forEach(r => counts[r.phase]++);
  counts[0] += Math.max(0, total - recs.length);
  const base = Math.max(total, recs.length) || 1;
  const weighted = recs.reduce((s, r) => s + r.phase / MAX, 0) / base;
  return { counts, base, weighted, recs };
}

function renderKpis({ counts, base, weighted, recs }) {
  const day = recs.filter(r => Date.now() - new Date(r.at) < 864e5).length;
  const items = [[Math.round(weighted * 100) + "%", "Overall progress"], [counts[4], "Completed"], [counts[1] + counts[2] + counts[3], "In progress"], [counts[0], "Not started"], [day, "Updated last 24 h"]];
  $("kpis").replaceChildren(...items.map(([v, l]) => el("div", { className: "kpi" }, el("b", {}, String(v)), el("span", {}, l))));
}

function renderDonut({ counts, base, weighted }) {
  let acc = 0;
  const stops = PHASES.map(([, c], i) => { const from = acc / base * 100; acc += counts[i]; return `${c} ${from}% ${acc / base * 100}%`; });
  $("donut").style.background = `conic-gradient(${stops.join(",")})`;
  $("donutPct").textContent = Math.round(weighted * 100) + "%";
  $("legend").replaceChildren(...PHASES.map(([n, c], i) => el("div", { className: "row" }, el("span", { className: "dot", style: { background: c } }), el("span", { className: "n" }, n), el("span", { className: "c" }, `${counts[i]} · ${Math.round(counts[i] / base * 100)}%`))));
}

function renderTrend({ base }) {
  const rows = [...history].sort((a, b) => a.at.localeCompare(b.at));
  const state = new Map(), byDay = new Map();
  let sum = 0;
  for (const r of rows) {
    sum += (r.phase - (state.get(r.guid) ?? 0)) / MAX;
    state.set(r.guid, r.phase);
    byDay.set(new Date(r.at).toLocaleDateString([], { day: "2-digit", month: "short" }), sum / base * 100);
  }
  const pts = [...byDay];
  if (!pts.length) return $("trend").replaceChildren(el("div", { className: "muted" }, "No history yet."));
  const W = 520, H = 170, l = 34, r = 10, t = 10, b = 24;
  const x = i => pts.length === 1 ? W / 2 : l + i * (W - l - r) / (pts.length - 1);
  const y = v => t + (100 - v) / 100 * (H - t - b);
  const grid = [0, 50, 100].map(v => `<line x1="${l}" x2="${W - r}" y1="${y(v)}" y2="${y(v)}" stroke="#e5e7eb"/><text x="${l - 6}" y="${y(v) + 4}" text-anchor="end">${v}%</text>`).join("");
  const line = pts.length > 1 ? `<polyline fill="none" stroke="#1d4ed8" stroke-width="2" points="${pts.map(([, v], i) => `${x(i)},${y(v)}`).join(" ")}"/>` : "";
  const dots = pts.map(([d, v], i) => `<circle cx="${x(i)}" cy="${y(v)}" r="4" fill="#1d4ed8"><title>${d}: ${v.toFixed(1)}%</title></circle>`).join("");
  const labels = pts.map(([d], i) => (pts.length <= 7 || i % Math.ceil(pts.length / 7) === 0) ? `<text x="${x(i)}" y="${H - 6}" text-anchor="middle">${d}</text>` : "").join("");
  $("trend").innerHTML = `<svg viewBox="0 0 ${W} ${H}" width="100%">${grid}${line}${dots}${labels}</svg>`;
}

function renderClasses() {
  if (!meta) return $("classes").replaceChildren(el("div", { className: "muted" }, "Reading element data…"));
  const map = new Map();
  for (const { guid, cls } of meta.values()) {
    const c = map.get(cls) ?? PHASES.map(() => 0);
    c[progress[guid]?.phase ?? 0]++;
    map.set(cls, c);
  }
  const rows = [...map].map(([cls, c]) => ({ cls, c, n: c.reduce((a, b) => a + b, 0), pct: c.reduce((s, v, i) => s + v * i / MAX, 0) })).sort((a, b) => b.n - a.n).slice(0, 12);
  $("classes").replaceChildren(table(["Type", "Count", "Phases", "Progress"],
    rows.map(r => [r.cls, el("span", {}, String(r.n)), stackBar(r.c, r.n), Math.round(r.pct / r.n * 100) + "%"])));
}

function renderRecent() {
  const rows = Object.entries(progress).sort((a, b) => b[1].at.localeCompare(a[1].at)).slice(0, 6);
  $("recent").replaceChildren(rows.length ? table(["Element", "Phase", "When"], rows.map(([g, r]) => [nameOf(g), badge(r.phase), fmt(r.at)])) : el("div", { className: "muted" }, "No updates yet."));
}

function renderElements() {

  }).sort((a, b) => b[1].at.localeCompare(a[1].at));
  $("elCount").textContent = `${rows.length} elements${rows.length > 200 ? " (showing 200)" : ""}`;

}

}

const dependents = id => {
  const out = new Set([id]);
  for (let grew = true; grew;) {
    grew = false;
    for (const t of schedule) if (!out.has(t.id) && (t.deps || []).some(d => out.has(d))) { out.add(t.id); grew = true; }
  }
  return out;
};

const STATUS = { todo: ["Not started", "#9ca3af"], active: ["In progress", "#2563eb"], late: ["Behind", "#ea580c"], done: ["Complete", "#16a34a"] };
function taskStatus(t) {
  const p = taskProgress(t), len = day(t.end) - day(t.start) + 1, today = Math.floor(Date.now() / 864e5);
  const planned = Math.round(Math.min(1, Math.max(0, (today - day(t.start) + 1) / len)) * 100);
  const status = p >= 100 ? "done" : planned > 0 && (today > day(t.end) || p < planned - 10) ? "late" : p > 0 ? "active" : "todo";
  return { p, planned, status };
}

const utcDay = d => new Date(d * 864e5).toISOString().slice(0, 10);

// finish-to-start, no lags; a task's own start acts as "start no earlier than"
function cpm() {
  const t0 = day(minStart());
  const nodes = new Map(schedule.map(t => [t.id, { t, d: day(t.end) - day(t.start) + 1, preds: (t.deps || []).filter(p => schedule.some(x => x.id === p)) }]));
  const order = [], seen = new Set();
  const visit = n => { if (seen.has(n.t.id)) return; seen.add(n.t.id); n.preds.forEach(p => visit(nodes.get(p))); order.push(n); };
  nodes.forEach(visit);
  for (const n of order) {
    n.es = Math.max(day(n.t.start) - t0, ...n.preds.map(p => nodes.get(p).ef));
    n.ef = n.es + n.d;
  }
  const end = Math.max(...order.map(n => n.ef));
  order.forEach(n => (n.lf = end));
  for (const n of [...order].reverse()) {
    n.ls = n.lf - n.d;
    n.float = n.ls - n.es;
    n.preds.forEach(p => { const q = nodes.get(p); q.lf = Math.min(q.lf, n.ls); });
  }
  return { nodes, t0, end };
}

    taskStatus(t).planned + "%", taskStatus(t).p + "%",
    el("span", { className: "badge", style: { background: STATUS[taskStatus(t).status][1], color: "#fff" } }, STATUS[taskStatus(t).status][0])])));

  if (box && scroll != null && !toToday) requestAnimationFrame(() => (box.scrollLeft = scroll));

}

const SIM_STYLE = { hidden: { visible: false }, active: { visible: true, color: "#ff8800" }, done: { visible: true, color: "#00cc00" } };

async function colourByDate() {
  const d = day(minStart()) + Number($("tSlide").value);
  $("tDate").textContent = utcDay(d);
  if (!simState.size) await api.viewer.setObjectState(undefined, { color: "#cccccc", visible: "reset" });
  for (const t of schedule) {
    const next = d < day(t.start) ? "hidden" : d > day(t.end) ? "done" : "active";
    if (simState.get(t.id) === next) continue;
    simState.set(t.id, next);
    const ids = await runtimeOf(t);
    if (ids.length) await api.viewer.setObjectState({ modelObjectIds: [{ modelId, objectRuntimeIds: ids }] }, SIM_STYLE[next]);
  }
}

async function colour() {
  const guids = Object.keys(progress);
  simState.clear();
  phaseByRuntime = new Map();
  await api.viewer.setObjectState(undefined, { color: "#cccccc" });
  if (guids.length) {
    const runtime = await api.viewer.convertToObjectRuntimeIds(modelId, guids);
    runtime.forEach((id, i) => id != null && phaseByRuntime.set(id, progress[guids[i]].phase));
    for (let ph = 1; ph < PHASES.length; ph++) {
      const ids = runtime.filter((id, i) => id != null && progress[guids[i]].phase === ph);
      if (ids.length) await api.viewer.setObjectState({ modelObjectIds: [{ modelId, objectRuntimeIds: ids }] }, { color: PHASES[ph][1] });
    }
  }
  await applyFilter();
}

async function applyFilter() {
  await api.viewer.setObjectState(undefined, { visible: "reset" });
  if (shown.size === PHASES.length) return;
  const hidden = PHASES.map(() => []);
  allIds.forEach(id => hidden[phaseByRuntime.get(id) ?? 0].push(id));
  for (let ph = 0; ph < PHASES.length; ph++) {
    if (!shown.has(ph) && hidden[ph].length) await api.viewer.setObjectState({ modelObjectIds: [{ modelId, objectRuntimeIds: hidden[ph] }] }, { visible: false });
  }
}

async function loadMeta() {
  const m = new Map();
  const STEP = 250;
  for (let i = 0; i < allIds.length; i += STEP) {
    const ids = allIds.slice(i, i + STEP);
    const [guids, props] = await Promise.all([api.viewer.convertToObjectIds(modelId, ids), api.viewer.getObjectProperties(modelId, ids)]);
    const byId = new Map(props.map(p => [Number(p.id), p]));
    ids.forEach((id, j) => {
      const p = byId.get(Number(id));
      m.set(id, { guid: guids[j], cls: typeof p?.class === "string" ? p.class : p?.class?.name || "Unclassified", name: typeof p?.product?.name === "string" ? p.product.name : "" });
    });
    status(`Reading element data ${Math.min(i + STEP, allIds.length)}/${allIds.length}`);
  }
  meta = m;
  guidMeta = new Map([...m.values()].map(v => [v.guid, v]));
  classes = [...new Set([...m.values()].map(v => v.cls))].sort();
  $("classSel").replaceChildren(...classes.map(c => el("option", { value: c }, c)));
  $("applyClass").disabled = !classes.length;
  openEditor(schedule.find(t => t.id === editId));
  render();
  status("Model loaded");
}

async function showSelection() {
  const sel = await api.viewer.getSelection();
  selIds = sel.filter(s => s.modelId === modelId).flatMap(s => s.objectRuntimeIds ?? []);
  $("applySel").disabled = !selIds.length;
  $("applySel").textContent = selIds.length > 1 ? `Apply to ${selIds.length} selected` : "Apply to selection";
  if (!selIds.length) { $("selInfo").textContent = "Select an element in the model."; return; }
  const [guid] = await api.viewer.convertToObjectIds(modelId, [selIds[0]]);
  const m = guidMeta.get(guid);
  const rec = progress[guid];
  $("selInfo").textContent = [m?.name || m?.cls || "Element", selIds.length > 1 ? `+ ${selIds.length - 1} more selected` : "", `GUID ${guid}`,
    rec ? `Phase: ${PHASES[rec.phase][0]}` : "No phase recorded yet.", rec ? `Recorded: ${fmt(rec.at)}` : ""].filter(Boolean).join("\n");
  if (rec) $("phase").value = rec.phase;
}

async function waitForObjects() {
  for (let i = 0; i < 60; i++) {
    const all = await api.viewer.getObjects();
    if (all[0]?.objects.length) return all;
    await new Promise(r => setTimeout(r, 1000));
  }
  throw new Error("model did not load in 60 s");
}

$("tAuto").onclick = guard(async () => {
  if (!meta) throw new Error("model data is still loading");
  if (schedule.length && !confirm("Replace the current schedule with one generated from the model?")) return;
  const byCls = new Map();
  for (const { guid, cls } of meta.values()) byCls.set(cls, [...(byCls.get(cls) ?? []), guid]);
  const stageOf = cls => { const i = STAGES.findIndex(re => re.test(cls.toLowerCase())); return i < 0 ? STAGES.length : i; };
  const start = utcDay(Math.floor(Date.now() / 864e5));
  const stages = new Map();
  const tasks = [...byCls].sort((a, b) => a[0].localeCompare(b[0])).map(([cls, guids]) => {
    const d = Math.max(2, Math.ceil(guids.length / 10));
    const t = { id: "a" + cls, name: `${cls} (${guids.length})`, start, end: utcDay(day(start) + d - 1), guids, deps: [] };
    const s = stageOf(cls);
    stages.set(s, [...(stages.get(s) ?? []), t]);
    return [s, t];
  });
  const prior = s => [...stages].filter(([k]) => k < s).sort((a, b) => b[0] - a[0])[0]?.[1] ?? [];
  schedule = tasks.map(([s, t]) => ({ ...t, deps: prior(s).map(p => p.id) }));
  const { nodes, t0 } = cpm();
  await saveSchedule(schedule.map(t => { const n = nodes.get(t.id); return { ...t, start: utcDay(t0 + n.es), end: utcDay(t0 + n.ef - 1) }; }));
});
